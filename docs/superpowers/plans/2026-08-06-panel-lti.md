# Panel de conexión con Moodle — plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Que un admin de Encuestum pueda conectar un Moodle, ver qué encuestas se usan
desde ahí y desconectarlo, todo desde el panel — y que un docente que agrega la actividad
sin usar "Seleccionar contenido" no deje a sus alumnos con un 404.

**Architecture:** No se toca la mecánica LTI, que ya funciona de punta a punta contra
Moodle 5.0.2 real. Se agregan endpoints de lectura sobre las tablas que ya existen
(`lti_platforms`, `lti_resource_links`), dos columnas nuevas para poder mostrar algo
legible y para el anonimato, y una sección en la página de Integraciones. El único cambio
de comportamiento en el flujo LTI es el fallback del selector cuando la actividad no tiene
encuesta asignada.

**Tech Stack:** FastAPI, SQLModel, Alembic, Next.js (App Router), Tailwind, pytest.

## Global Constraints

- Todo detrás de `LTI_ENABLED` (default apagado). Con el flag apagado, `/lti/*` y
  `/api/v1/lti/*` responden 404 y nada del comportamiento actual cambia.
- Los endpoints del panel van en `admin_router` (prefijo `/api/v1/lti`), que ya existe en
  `backend/app/routers/lti.py:1031`, y requieren **sesión + rango admin de organización**
  — el mismo chequeo que ya usa `create_platform` (`ROLE_RANK.get(ctx.role, 0) < ROLE_RANK[ROLE_ADMIN]` → 403).
- **Aislamiento entre organizaciones**: cada consulta filtra por `platform.org_id == ctx.org.id`.
  Nunca se confía en el id que viene por la URL.
- Comentarios y docstrings en español, como el resto de `backend/app/`.
- Migraciones Alembic **idempotentes** (patrón de `backend/alembic/versions/0019_lti.py`:
  inspeccionar antes de crear). La última migración es `0019_lti`; la nueva es `0020_lti_panel`.
- Frontend: llamadas con `getApiUrl()` y `credentials: "include"`, errores como
  `Error(detail)` — el patrón de `frontend/utils/webhooks.ts`.
- Textos de interfaz en el diccionario i18n, en los 7 idiomas de
  `frontend/lib/i18n-dicts/`. Nada hardcodeado.
- Cada tarea termina con `pytest` verde y un commit.

async def _plataforma_de_la_org(session: AsyncSession, ctx: OrgContext, pid: uuid.UUID) -> LtiPlatform:
    """La plataforma pedida, sólo si pertenece a la organización activa.

    Una plataforma ajena devuelve 404, no 403: quién tiene qué LMS conectado no
    es información que corresponda filtrar a otra organización."""
    p = await session.get(LtiPlatform, pid)
    if p is None or p.org_id != ctx.org.id:
        raise HTTPException(status_code=404, detail="Plataforma no encontrada.")
    return p


def _exigir_admin(ctx: OrgContext) -> None:
    if ROLE_RANK.get(ctx.role, 0) < ROLE_RANK[ROLE_ADMIN]:
        raise HTTPException(status_code=403, detail="Necesitás ser admin de la organización.")


@admin_router.get("/platforms", dependencies=[Depends(require_lti)])
async def list_platforms(
    ctx: OrgContext = Depends(current_context),
    session: AsyncSession = Depends(get_session),
) -> list[dict]:
    """Los LMS conectados a esta organización, con cuántas actividades y
    respuestas tiene cada uno."""
    _exigir_admin(ctx)
    plataformas = (
        await session.scalars(
            select(LtiPlatform)
            .where(LtiPlatform.org_id == ctx.org.id)
            .order_by(LtiPlatform.created_at.desc())
        )
    ).all()

    salida = []
    for p in plataformas:
        links = (
            await session.scalars(
                select(LtiResourceLink).where(LtiResourceLink.platform_id == p.id)
            )
        ).all()
        ids = [l.id for l in links]
        respuestas = 0
        if ids:
            respuestas = (
                await session.scalar(
                    select(func.count())
                    .select_from(SurveyResponse)
                    .where(SurveyResponse.lti_link_id.in_(ids))
                )
            ) or 0
        salida.append({
            "id": str(p.id),
            "issuer": p.issuer,
            "name": p.name,
            "created_at": p.created_at.isoformat(),
            "activities": len(links),
            "responses": respuestas,
        })
    return salida


@admin_router.get("/platforms/{platform_id}/links", dependencies=[Depends(require_lti)])
async def list_platform_links(
    platform_id: uuid.UUID,
    ctx: OrgContext = Depends(current_context),
    session: AsyncSession = Depends(get_session),
) -> list[dict]:
    """Qué encuesta se usa en qué curso, y cuántas respuestas llegaron por ahí.

    Es lo que contesta "¿por qué no me llega la nota?": cero respuestas señala a
    Moodle, respuestas sin nota señalan acá."""
    _exigir_admin(ctx)
    p = await _plataforma_de_la_org(session, ctx, platform_id)

    links = (
        await session.scalars(
            select(LtiResourceLink)
            .where(LtiResourceLink.platform_id == p.id)
            .order_by(LtiResourceLink.created_at.desc())
        )
    ).all()

    salida = []
    for l in links:
        encuesta = await session.get(Survey, l.survey_id)
        total = (
            await session.scalar(
                select(func.count())
                .select_from(SurveyResponse)
                .where(SurveyResponse.lti_link_id == l.id)
            )
        ) or 0
        ultima = await session.scalar(
            select(func.max(SurveyResponse.submitted_at)).where(
                SurveyResponse.lti_link_id == l.id
            )
        )
        salida.append({
            "id": str(l.id),
            "survey_id": str(l.survey_id),

Agregar en `backend/app/routers/lti.py`, junto a los otros endpoints del selector:

```python
class LinkReturn(BaseModel):
    link: str
    survey_id: uuid.UUID
    anonymous: bool = False


@router.post("/select/link", dependencies=[Depends(require_lti)])
async def select_link(
    payload: LinkReturn,
    session: AsyncSession = Depends(get_session),
) -> dict:
    """Vincula la encuesta elegida a la actividad desde la que se llegó y
    devuelve a dónde ir. A diferencia de `/select/return`, acá no se le firma
    nada a la plataforma: el vínculo se guarda de este lado."""
    datos = read_purpose_token(LINK_PURPOSE, payload.link or "")
    if not datos:
        raise HTTPException(status_code=400, detail="Sesión de selección vencida.")

    platform = await session.get(LtiPlatform, uuid.UUID(datos["platform_id"]))
    if platform is None:
        raise HTTPException(status_code=400, detail="Plataforma LTI no registrada.")

    survey = await session.get(Survey, payload.survey_id)
    if survey is None or survey.deleted_at is not None or survey.org_id != platform.org_id:
        raise HTTPException(status_code=404, detail="Encuesta no encontrada.")

    link = LtiResourceLink(
        platform_id=platform.id,
        resource_link_id=datos["resource_link_id"],
        survey_id=survey.id,
        context_title=datos.get("context_title"),
        anonymous=payload.anonymous,
    )
    session.add(link)
    try:
        await session.commit()
    except IntegrityError:
        # Dos docentes eligiendo a la vez sobre la misma actividad: gana el
        # primero y el segundo sigue con ese vínculo.
        await session.rollback()
        link = (
            await session.scalars(
                select(LtiResourceLink).where(
                    LtiResourceLink.platform_id == uuid.UUID(datos["platform_id"]),
                    LtiResourceLink.resource_link_id == datos["resource_link_id"],
                )
            )
        ).first()
        if link is None:
            raise

    return {"redirect": f"/s/{survey.slug}"}
```

async function pedir<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(getApiUrl(path), { credentials: "include", ...init });
  if (!res.ok) {
    let detalle = "No se pudo completar la operación.";
    try {
      const d = await res.json();
